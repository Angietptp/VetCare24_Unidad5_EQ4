1"""Specification de la atención sin cita (D-06, R-06, NB-19; cierra G-06). 

2 

3Una mascota puede recibir atención si tiene una cita confirmada O si llegó como emergencia: 

4 

5 atencion_permitida = TieneCitaConfirmada OR TieneEmergenciaActiva 

6 

7Es una consulta (no cambia nada): la API la expone como evaluación, igual que L-01. 

8""" 

9 

10from __future__ import annotations 

11 

12from collections.abc import Iterable 

13from dataclasses import dataclass 

14 

15from vetcare24.domain._validacion import exigir_texto 

16from vetcare24.domain.cita import Cita, EstadoCita 

17from vetcare24.domain.emergencia import Emergencia 

18from vetcare24.domain.specification import Specification 

19 

20 

21@dataclass(frozen=True) 

22class MascotaParaAtencion: 

23 """Una mascota con sus citas y sus emergencias (solo las suyas).""" 

24 

25 mascota_id: str 

26 citas: tuple[Cita, ...] 

27 emergencias: tuple[Emergencia, ...] 

28 

29 @classmethod 

30 def de(cls, mascota_id: str, citas: Iterable[Cita], 

31 emergencias: Iterable[Emergencia]) -> MascotaParaAtencion: 

32 """Arma el contexto quedándose solo con lo que es de esa mascota.""" 

33 exigir_texto(mascota_id, "MASCOTA_REQUERIDA", "La mascota") 

34 return cls( 

35 mascota_id, 

36 tuple(c for c in citas if c.mascota_id == mascota_id), 

37 tuple(e for e in emergencias if e.mascota_id == mascota_id), 

38 ) 

39 

40 

41Regla = Specification[MascotaParaAtencion] 

42 

43 

44class TieneCitaConfirmada(Regla): 

45 """Una cita programada para la mascota. SOLICITADA aún no está programada (CitaAgendada se 

46 emite al confirmar, I-01); REALIZADA y CANCELADA ya no dan pie a una atención nueva.""" 

47 

48 codigo = "SIN_CITA_CONFIRMADA" 

49 

50 def es_satisfecha_por(self, c: MascotaParaAtencion) -> bool: 

51 return any(cita.estado is EstadoCita.CONFIRMADA for cita in c.citas) 

52 

53 

54class TieneEmergenciaActiva(Regla): 

55 codigo = "SIN_EMERGENCIA_ACTIVA" 

56 

57 def es_satisfecha_por(self, c: MascotaParaAtencion) -> bool: 

58 return any(emergencia.esta_activa for emergencia in c.emergencias) 

59 

60 

61def atencion_permitida() -> Regla: 

62 """R-06: TieneCitaConfirmada OR TieneEmergenciaActiva.""" 

63 return TieneCitaConfirmada() | TieneEmergenciaActiva()
