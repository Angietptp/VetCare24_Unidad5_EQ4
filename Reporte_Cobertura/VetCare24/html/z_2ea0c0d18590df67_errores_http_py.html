1"""Traducción de errores a respuestas HTTP (D-13, H-07) con mensajes para el cliente. 

2 

3Función pura: no depende de FastAPI, así se prueba sin levantar el servidor. 

4Todas las respuestas de error tienen la misma forma: 

5 {"error": {"codigo": "...", "mensaje": "...", "detalles": [...], "motivos": [...]}} 

6`codigo` es corto y estable (para programas); `mensaje` es una frase en español pensada para 

7mostrarse tal cual a quien usa el sistema. `detalles` señala los campos inválidos de una 

8solicitud mal formada. `motivos` trae los códigos de las reglas que se incumplieron (por 

9ejemplo ACTOR_INACTIVO), para que un programa sepa qué falló sin leer la frase. 

10"""

104def cuerpo_de_error( 

105 codigo: str, mensaje: str, detalles: list[Any] | None = None, 

106 motivos: tuple[str, ...] = (), 

107) -> dict: 

108 return {"error": { 

109 "codigo": codigo, "mensaje": mensaje, "detalles": detalles or [], "motivos": list(motivos), 

110 }}

115def _oracion(texto: str) -> str: 

116 texto = texto.strip() 

117 texto = texto[:1].upper() + texto[1:] 

118 return texto if texto.endswith((".", "!", "?")) else texto + "." 

119 

120 

121def _motivo(motivo: str) -> str: 

122 codigo, _, servicio = motivo.partition(":") 

123 frase = MOTIVOS.get(codigo) 

124 if frase is None: 

125 return codigo.lower().replace("_", " ") 

126 return frase.format(x=servicio or "indicado") 

127 

128 

129def _invariante(error: InvarianteViolada) -> str: 

130 plantilla = INVARIANTES.get(error.codigo) 

131 if plantilla is None: 

132 return _oracion(error.mensaje) 

133 return plantilla.format(x=error.mensaje.split(" ")[0]) # el mensaje empieza con el id

136def _articulo(tipo: str, indefinido: bool = False) -> str: 

137 definido, indef = ARTICULOS.get(tipo, ("el", "un")) 

138 return indef if indefinido else definido

143def detalle_de_validacion(error: dict) -> dict: 

144 """Convierte un error de validación de FastAPI en un campo y una frase en español.""" 

145 loc = [str(parte) for parte in error.get("loc", ())] 

146 if loc and loc[0] in ("body", "query", "path", "header"): 146 ↛ 148line 146 didn't jump to line 148 because the condition on line 146 was always true

147 loc = loc[1:] 

148 campo = ".".join(loc) or "solicitud" 

149 tipo = error.get("type", "") 

150 if tipo == "json_invalid": 

151 return {"campo": "cuerpo", "mensaje": "El cuerpo de la solicitud no es un JSON válido."} 

152 if tipo == "missing": 

153 frase = f"Falta el campo «{campo}»." 

154 elif tipo == "extra_forbidden": 

155 frase = f"El campo «{campo}» no se esperaba." 

156 elif tipo in ("enum", "literal_error"): 

157 permitidos = (error.get("ctx") or {}).get("expected") 

158 extra = f": {str(permitidos).replace(' or ', ' o ')}" if permitidos else "" 

159 frase = f"El campo «{campo}» tiene un valor no permitido; debe ser uno de{extra}." 

160 else: 

161 frase = f"El campo «{campo}» {TIPOS_DE_CAMPO.get(tipo, 'tiene un valor no válido')}." 

162 return {"campo": campo, "mensaje": frase}

165def solicitud_invalida(errores: list[dict]) -> tuple[int, dict]: 

166 detalles = [detalle_de_validacion(e) for e in errores] 

167 return 400, cuerpo_de_error( 

168 "SOLICITUD_INVALIDA", "La solicitud no es válida. Revisa los campos indicados.", detalles)

173def traducir(error: Exception) -> tuple[int, dict] | None: 

174 """Devuelve (código HTTP, cuerpo) para un error de negocio; None si no es de negocio.""" 

175 if isinstance(error, RolNoIndicado): 

176 return 401, cuerpo_de_error( 

177 "ROL_REQUERIDO", 

178 f"Falta indicar tu rol. Envía el encabezado X-Rol con uno de estos valores: {ROLES}.") 

179 if isinstance(error, RolDesconocido): 

180 return 400, cuerpo_de_error( 

181 "ROL_DESCONOCIDO", f"El rol «{error.valor}» no existe. Usa {ROLES}.") 

182 if isinstance(error, AccesoDenegado): 

183 rol = ROLES_LEGIBLES.get(error.rol, error.rol) 

184 accion = PERMISOS_LEGIBLES.get(error.permiso, error.permiso) 

185 return 403, cuerpo_de_error( 

186 "ACCESO_DENEGADO", f"El rol {rol} no tiene permiso para {accion}.") 

187 if isinstance(error, NoEncontrado): 

188 return 404, cuerpo_de_error( 

189 "NO_ENCONTRADO", 

190 f"No se encontró {_articulo(error.tipo)} {error.tipo} «{error.identificador}».") 

191 if isinstance(error, YaExiste): 

192 return 409, cuerpo_de_error( 

193 "YA_EXISTE", 

194 f"Ya existe {_articulo(error.tipo, True)} {error.tipo} " 

195 f"con el identificador «{error.identificador}».") 

196 if isinstance(error, TransicionInvalida): 

197 pista = PISTAS.get((error.estado, error.accion), "") 

198 return 409, cuerpo_de_error( 

199 "TRANSICION_INVALIDA", 

200 f"No se puede {ACCIONES_LEGIBLES.get(error.accion, error.accion)} " 

201 f"{_articulo(error.entidad, True)} {error.entidad} que está " 

202 f"{ESTADOS_LEGIBLES.get(error.estado, error.estado.lower())}.{pista}", 

203 motivos=(error.motivo,)) 

204 if isinstance(error, PrecondicionNoCumplida): 

205 razones = "; ".join(_motivo(m) for m in error.motivos) 

206 mensaje = (f"No se puede {error.accion} {_articulo(error.entidad)} {error.entidad}: " 

207 f"{razones}.") 

208 # H-07: falta de capacidad es un conflicto (409); las demás precondiciones, 422. 

209 if any(m.startswith(CAPACIDAD) for m in error.motivos): 

210 return 409, cuerpo_de_error(CAPACIDAD, mensaje, motivos=error.motivos) 

211 return 422, cuerpo_de_error("PRECONDICION_NO_CUMPLIDA", mensaje, motivos=error.motivos) 

212 if isinstance(error, InvarianteViolada): 

213 return 422, cuerpo_de_error(error.codigo, _invariante(error)) 

214 if isinstance(error, (ErrorDeDominio, ErrorDeAplicacion)): 

215 return 422, cuerpo_de_error("REGLA_DE_NEGOCIO", _oracion(str(error))) 

216 return None

219def error_http(estado: int) -> tuple[int, dict]: 

220 """Errores que FastAPI genera solo (ruta inexistente, método no permitido) con el mismo 

221 formato que el resto, para que el cliente siempre lea `error.codigo`.""" 

222 if estado == 404: 

223 return 404, cuerpo_de_error("RUTA_NO_ENCONTRADA", "La dirección solicitada no existe.") 

224 if estado == 405: 

225 return 405, cuerpo_de_error( 

226 "METODO_NO_PERMITIDO", "Ese método HTTP no está permitido en esta dirección.") 

227 return estado, cuerpo_de_error("ERROR_HTTP", "No se pudo atender la solicitud.")
