1"""Specifications sobre la Cita (Etapa 4). 

2 

3Todas evalúan un `CitaEnContexto`: la cita junto con lo que hace falta para juzgarla 

4(su propietario, el catálogo y la capacidad disponible). Los códigos de motivo coinciden con 

5los del oráculo. 

6 

7Origen de cada regla: 

8- BRD: ActorActivo, DetalleNoVacio, Autorizada, ServiciosHabilitados, CapacidadSuficiente, 

9 estados que permiten cancelar o modificar (NB-08, R-09, R-14, D-07, A-06). 

10- Extensión del paquete (D-05): PropietarioPreferente, TotalMinimo, PromocionActiva. 

11"""

28def disponibilidad_inicial(catalogo: Mapping[str, ServicioVeterinario]) -> dict[str, int]: 

29 """Capacidad completa de cada servicio, antes de confirmar ninguna cita.""" 

30 return {servicio_id: s.capacidad for servicio_id, s in catalogo.items()}

33@dataclass(frozen=True) 

34class CitaEnContexto: 

35 cita: Cita 

36 propietario: Propietario 

37 catalogo: Mapping[str, ServicioVeterinario] 

38 disponible: Mapping[str, int] = field(default_factory=dict) 

39 

40 def __post_init__(self) -> None: 

41 if self.cita.propietario_id != self.propietario.propietario_id: 

42 raise InvarianteViolada( 

43 "PROPIETARIO_INCORRECTO", 

44 f"{self.cita.cita_id} es de {self.cita.propietario_id}, " 

45 f"no de {self.propietario.propietario_id}", 

46 )

102class ServiciosHabilitados(Regla): 

103 """Todos los servicios de la cita existen en el catálogo y están habilitados.""" 

104 

105 def _problemas(self, c: CitaEnContexto) -> Iterator[str]: 

106 for linea in c.cita.lineas: 

107 servicio = c.catalogo.get(linea.servicio_id) 

108 if servicio is None: 

109 yield f"SERVICIO_INEXISTENTE:{linea.servicio_id}" 

110 elif not servicio.habilitado: 

111 yield f"SERVICIO_DESHABILITADO:{linea.servicio_id}" 

112 

113 def es_satisfecha_por(self, c: CitaEnContexto) -> bool: 

114 return next(self._problemas(c), None) is None 

115 

116 def motivos_de_incumplimiento(self, c: CitaEnContexto) -> tuple[str, ...]: 

117 return tuple(dict.fromkeys(self._problemas(c)))

120class CapacidadSuficiente(Regla): 

121 """La demanda de cada servicio cabe en la capacidad disponible.""" 

122 

123 def _faltantes(self, c: CitaEnContexto) -> Iterator[str]: 

124 for servicio_id, pedido in c.cita.demanda_por_servicio().items(): 

125 if pedido > c.disponible.get(servicio_id, 0): 

126 yield f"CAPACIDAD_INSUFICIENTE:{servicio_id}" 

127 

128 def es_satisfecha_por(self, c: CitaEnContexto) -> bool: 

129 return next(self._faltantes(c), None) is None 

130 

131 def motivos_de_incumplimiento(self, c: CitaEnContexto) -> tuple[str, ...]: 

132 return tuple(dict.fromkeys(self._faltantes(c)))

149def puede_procesarse() -> Regla: 

150 """D-03: ActorActivo AND DetalleNoVacio AND Autorizada.""" 

151 return ActorActivo() & DetalleNoVacio() & Autorizada()

154def puede_confirmarse() -> Regla: 

155 """D-12: elegibilidad AND servicios habilitados AND capacidad suficiente.""" 

156 return puede_procesarse() & ServiciosHabilitados() & CapacidadSuficiente()

159def beneficio_de_referencia(umbral: Money = UMBRAL_BENEFICIO) -> Regla: 

160 """D-05: PropietarioPreferente AND TotalMinimo AND NOT PromocionActiva.""" 

161 sin_promocion = Not(PromocionActiva(), "PROMOCION_ACTIVA") 

162 return PropietarioPreferente() & TotalMinimo(umbral) & sin_promocion

165def esta_finalizada() -> Regla: 

166 """D-07: REALIZADA OR CANCELADA (estados terminales).""" 

167 return EnEstado(EstadoCita.REALIZADA) | EnEstado(EstadoCita.CANCELADA)

170def puede_cancelarse() -> Regla: 

171 """D-07 / NB-08: se cancela mientras la cita no esté finalizada.""" 

172 return Not(esta_finalizada(), "CITA_FINALIZADA")

175def puede_modificarse() -> Regla: 

176 """A-06 / NB-08: se modifica mientras la cita no esté finalizada.""" 

177 return Not(esta_finalizada(), "CITA_FINALIZADA")
